}}{{ partial "metadata/row.html" (dict "label" "Captured" "value" (.Format "2006-01-02 15:04")) }}{{ end }}{{ end -}} {{- end -}} {{- $make := $exif.Make | default "" -}} {{- $model := $exif.Model | $lens) " " }}{{ end -}} {{- partial "metadata/row.html" (dict "label" "Lens" "value" $lens) -}} {{- end -}} {{- with $exif.FocalLength -}} {{- $focal := printf "%.3g mm" (partial
