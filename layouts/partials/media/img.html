{{/* A responsive, progressively loading . Expects a dict: image - the image resource widths - candidate widths in px (default 320, 640, 1280) sizes - the sizes attribute (default 100vw) alt - alt text (default empty) original - true adds the lossless source as the largest candidate focus - [x y] percentages for object-position eager - true loads at once, with high priority (above the fold) The browser picks the smallest candidate that is sharp enough for the slot and the screen density. A 24 px blurred copy is inlined as the background, so the slot is never blank while the pixels arrive. SVGs and images narrower than every width come back unchanged. */}} {{- $img := .image -}} {{- $alt := .alt | default "" -}} {{- $loading := cond (.eager | default false) "eager" "lazy" -}} {{- $style := "" -}} {{- with .focus -}}{{- $style = printf "object-position:%v%% %v%%;" (index . 0) (index . 1) -}}{{- end -}} {{- if or (ne $img.ResourceType "image") (eq $img.MediaType.SubType "svg") -}} {{- else -}} {{- $widths := .widths | default (slice 320 640 1280) -}} {{- $candidates := slice -}} {{- $src := $img.RelPermalink -}} {{- range $widths -}} {{- if lt . $img.Width -}} {{- $resized := $img.Resize (printf "%dx webp q80 picture" .) -}} {{- $candidates = $candidates | append (printf "%s %dw" $resized.RelPermalink $resized.Width) -}} {{- $src = $resized.RelPermalink -}} {{- end -}} {{- end -}} {{- if .original -}} {{- $candidates = $candidates | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- end -}} {{- $tiny := $img.Resize "24x webp q30 picture" -}} {{- $style = printf "%sbackground-image:url(data:image/webp;base64,%s)" $style ($tiny.Content | base64Encode) -}} {{- end -}}
